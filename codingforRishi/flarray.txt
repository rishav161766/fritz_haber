class Solution {
public:
    vector<int> searchRange(vector<int>& nums, int target) {
        vector<int>ans(2,-1);
        int n=nums.size();
        int low=0,high=n-1;
        while(low<=high){
            int mid=low+(high-low)/2;
             if(nums[mid]==target){
                if(mid==0 ||nums[mid-1]!=target){
                    ans[0]=mid;
                    break;
                }
                high=mid-1;
            }
            if(nums[mid]>target) high=mid-1;
            if(nums[mid]<target) low=mid+1;
           
        }
        
        int l=0,h=n-1;
        while(l<=h){
            int m=l+(h-l)/2;
            if(nums[m]==target){
                if( m==n-1 ||nums[m+1]!=target){
                   ans[1]=m;
                    break;
                }
                l=m+1;
            }
            if(nums[m]>target) h=m-1;
            if(nums[m]<target) l=m+1;
            
        }
        
    return ans;
    }
};
