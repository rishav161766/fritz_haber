class Solution {
public:
int product(int n){
    int prod=1;
    while(n>0){
        int r=n%10;
        prod*=r;
        n/=10;
    }
    return prod;
}
    int smallestNumber(int n, int t) {
        for(int i=n;;i++){
            if(product(i)%t==0){
                return i;
            }
        }
    }
};
